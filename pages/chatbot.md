<!-- Load marked.js and DOMPurify to render sanitized markdown generated by Gemini -->

<script>
const RAG_API_URL = window.location.hostname.includes('netlify.app') || window.location.hostname === 'localhost'
  ? '/.netlify/functions/rag'
  : 'https://unknownhands.netlify.app/.netlify/functions/rag';

function appendLabel(parent, label) {
  const strong = document.createElement('strong');
  strong.textContent = label;
  parent.appendChild(strong);
}

function appendUserMessage(chatMessages, message) {
  const userDiv = document.createElement('div');
  userDiv.className = 'user-message';
  appendLabel(userDiv, 'You:');
  userDiv.appendChild(document.createTextNode(' ' + message));
  chatMessages.appendChild(userDiv);
}

function appendBotMessage(chatMessages, data) {
  const botDiv = document.createElement('div');
  botDiv.className = 'bot-message';
  appendLabel(botDiv, 'Assistant:');
  botDiv.appendChild(document.createElement('br'));

  const markdownDiv = document.createElement('div');
  markdownDiv.className = 'rendered-markdown';
  const answer = data.answer || "Sorry, I couldn't generate an answer.";
  const rendered = marked.parse(answer);
  markdownDiv.innerHTML = DOMPurify.sanitize(rendered);
  botDiv.appendChild(markdownDiv);

  if (Array.isArray(data.sources) && data.sources.length > 0) {
    const sourceDiv = document.createElement('div');
    sourceDiv.className = 'source-badges';
    appendLabel(sourceDiv, 'Referenced Manuscripts:');
    sourceDiv.appendChild(document.createElement('br'));
    data.sources.forEach((source) => {
      const badge = document.createElement('span');
      badge.className = 'source-badge';
      badge.textContent = source.title || source.id || 'Untitled source';
      sourceDiv.appendChild(badge);
    });
    botDiv.appendChild(sourceDiv);
  }

  chatMessages.appendChild(botDiv);
}

function appendErrorMessage(chatMessages, error) {
  const isRateLimit = error.message.includes("high traffic") || error.message.includes("rate limit");
  const errorDiv = document.createElement('div');
  errorDiv.className = 'bot-message';
  errorDiv.style.backgroundColor = isRateLimit ? '#fff3cd' : '#f8d7da';
  errorDiv.style.color = '#721c24';
  appendLabel(errorDiv, isRateLimit ? 'System Notice:' : 'Error:');
  errorDiv.appendChild(document.createTextNode(' ' + error.message));
  chatMessages.appendChild(errorDiv);
}

async function sendMessage() {
  const input = document.getElementById('user-input');
  const message = input.value.trim();
  if (!message) return;
  
  // Clear input
  input.value = '';
  
  const chatMessages = document.getElementById('chat-messages');
  
  appendUserMessage(chatMessages, message);
  
  // Show loading
  document.getElementById('loading-indicator').style.display = 'block';
  chatMessages.scrollTop = chatMessages.scrollHeight;
  
  try {
    const response = await fetch(RAG_API_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ query: message })
    });
    
    // Add explicitly handled error checking!
    if (!response.ok) {
      if (response.headers.get('content-type')?.includes('text/html')) {
        throw new Error("Server timeout or Netlify gateway error (Status: " + response.status + ")");
      }
      const errData = await response.json().catch(() => ({}));
      throw new Error(errData.error || "Failed to communicate with the semantic engine.");
    }

    const data = await response.json();
    
    document.getElementById('loading-indicator').style.display = 'none';
    
    appendBotMessage(chatMessages, data);
    
  } catch (error) {
    console.error(error);
    document.getElementById('loading-indicator').style.display = 'none';
    appendErrorMessage(chatMessages, error);
  }
  
  chatMessages.scrollTop = chatMessages.scrollHeight;
}
</script>
